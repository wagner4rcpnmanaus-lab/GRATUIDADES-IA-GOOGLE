const SHEET_ID = '16nPjDWqQktXvrxK6RsK4VBuDNOXN81wTNGgjYNSRiec';
const HEADERS = [
  'id',
  'protocolo',
  'escrevente',
  'motivo',
  'tipoCertidao',
  'parteInteressada',
  'telefone',
  'email',
  'data',
  'observacoes',
  'naoAtendido',
  'createdAt',
  'updatedAt',
];

function sheet_() {
  const ss = SpreadsheetApp.openById(SHEET_ID);
  const sh = ss.getSheets()[0];
  if (sh.getLastRow() === 0) sh.appendRow(HEADERS);
  const current = sh.getRange(1, 1, 1, HEADERS.length).getValues()[0];
  if (current.join('|') !== HEADERS.join('|')) {
    sh.getRange(1, 1, 1, HEADERS.length).setValues([HEADERS]);
  }
  // Mantém protocolo ("001") e data ("2026-08-18") como TEXTO,
  // impedindo que o Sheets converta em número ou objeto de data.
  sh.getRange(1, 2, sh.getMaxRows(), 1).setNumberFormat('@'); // protocolo
  sh.getRange(1, 9, sh.getMaxRows(), 1).setNumberFormat('@'); // data
  return sh;
}

function json_(payload) {
  return ContentService
    .createTextOutput(JSON.stringify(payload))
    .setMimeType(ContentService.MimeType.JSON);
}

function normalizeDate_(value, ss) {
  if (value instanceof Date) {
    return Utilities.formatDate(value, ss.getSpreadsheetTimeZone(), 'yyyy-MM-dd');
  }
  return String(value || '');
}

function doGet() {
  const ss = SpreadsheetApp.openById(SHEET_ID);
  const sh = sheet_();
  const values = sh.getDataRange().getValues();
  const rows = values.slice(1).filter(row => row.some(cell => String(cell).trim() !== ''));
  const records = rows.map(row => {
    const obj = {};
    HEADERS.forEach((h, i) => obj[h] = row[i]);
    obj.protocolo = String(obj.protocolo).padStart(3, '0');
    obj.data = normalizeDate_(obj.data, ss);
    obj.naoAtendido = String(obj.naoAtendido).toLowerCase() === 'true';
    obj.createdAt = Number(obj.createdAt) || Date.now();
    obj.updatedAt = Number(obj.updatedAt) || Date.now();
    return obj;
  });
  return json_({ ok: true, records });
}

function doPost(e) {
  const payload = JSON.parse(e.postData.contents || '{}');
  if (payload.action !== 'replaceAll') {
    return json_({ ok: false, error: 'Ação inválida.' });
  }

  const records = Array.isArray(payload.records) ? payload.records : [];
  const sh = sheet_();
  sh.clearContents();
  sh.getRange(1, 1, 1, HEADERS.length).setValues([HEADERS]);

  if (records.length > 0) {
    const rows = records.map(record => HEADERS.map(h => record[h] ?? ''));
    sh.getRange(2, 1, rows.length, HEADERS.length).setValues(rows);
  }

  return json_({ ok: true, count: records.length });
}
